cmake_minimum_required(VERSION 3.16)
project(JauCppMathExtension LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(cppmath OBJECT cppmath.cpp)
if(MSVC)
  target_compile_options(cppmath PRIVATE /O2 /GS- /GR- /EHs-c-)
else()
  target_compile_options(cppmath PRIVATE -O2 -fno-exceptions -fno-rtti)
endif()
