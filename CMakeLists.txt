cmake_minimum_required(VERSION 3.16)
project(Jau VERSION 0.8.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JAU_ENABLE_ASM_FASTOPS "Enable architecture-specific ASM fast paths" ON)
add_library(jaucore src/jau.cpp src/package.cpp)
target_include_directories(jaucore PUBLIC include)
if(MSVC)
  target_compile_options(jaucore PRIVATE /W4 /permissive-)
else()
  target_compile_options(jaucore PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(JAU_ENABLE_ASM_FASTOPS AND NOT MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8 AND (CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64"))
  enable_language(ASM)
  if(NOT MSVC)
    target_sources(jaucore PRIVATE src/fastops_x86_64.S)
    target_compile_definitions(jaucore PRIVATE JAU_HAVE_ASM_FASTOPS=1)
  endif()
endif()
add_executable(jauas src/jauas_main.cpp)
add_executable(jauld src/jauld_v2_main.cpp)
add_executable(jauc src/jauc_main.cpp)
target_link_libraries(jauc PRIVATE jaucore)
add_executable(jur src/jur_main.cpp)
target_link_libraries(jur PRIVATE jaucore)
add_dependencies(jauc jur jauas jauld)
add_custom_command(TARGET jauc POST_BUILD
  COMMAND $<TARGET_FILE:jauc> standalone ${CMAKE_SOURCE_DIR}/tools/jaupm.jau -o $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX} --runtime $<TARGET_FILE:jur> -I ${CMAKE_SOURCE_DIR}/stdlib
  COMMENT "Building JauPM standalone executable with Jau"
  VERBATIM)
add_executable(jau-setup installer/jau_setup.cpp)
if(WIN32)
  target_link_libraries(jau-setup PRIVATE advapi32)
endif()
enable_testing()
add_test(NAME jau_smoke COMMAND jauc run ${CMAKE_SOURCE_DIR}/examples/hello.jau)
add_test(NAME jau_functions COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/functions.jau)
add_test(NAME jau_imports COMMAND jauc run ${CMAKE_SOURCE_DIR}/examples/imports.jau -I ${CMAKE_SOURCE_DIR}/stdlib)
add_test(NAME jau_bootstrap_seed COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/bootstrap_seed.jau)
add_test(NAME jau_namespaces COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/namespaces.jau)
if(UNIX AND NOT APPLE)
  add_test(NAME jau_bootstrap_stage1 COMMAND sh ${CMAKE_SOURCE_DIR}/tests/bootstrap_stage1.sh $<TARGET_FILE:jauc> $<TARGET_FILE:jauas> ${CMAKE_SOURCE_DIR} linux-x86_64)
  add_test(NAME jau_package_manager COMMAND sh ${CMAKE_SOURCE_DIR}/tests/package_manager.sh $<TARGET_FILE:jauc> ${CMAKE_SOURCE_DIR})
  add_test(NAME jau_standalone COMMAND sh ${CMAKE_SOURCE_DIR}/tests/standalone.sh $<TARGET_FILE:jauc> $<TARGET_FILE:jur> ${CMAKE_SOURCE_DIR})
  add_test(NAME jau_standalone_args COMMAND sh ${CMAKE_SOURCE_DIR}/tests/standalone_args.sh $<TARGET_FILE:jauc> $<TARGET_FILE:jur> ${CMAKE_SOURCE_DIR})
  add_test(NAME jaupm_binary COMMAND sh ${CMAKE_SOURCE_DIR}/tests/jaupm_binary.sh $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX} $<TARGET_FILE:jauc> ${CMAKE_SOURCE_DIR})
  add_test(NAME jaupm_path COMMAND sh ${CMAKE_SOURCE_DIR}/tests/standalone_path.sh $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX})
  add_test(NAME jau_object_interop COMMAND sh ${CMAKE_SOURCE_DIR}/tests/object_interop.sh $<TARGET_FILE:jauc> ${CMAKE_SOURCE_DIR})
  add_test(NAME jau_native_cpp_link COMMAND sh ${CMAKE_SOURCE_DIR}/tests/native_link.sh $<TARGET_FILE:jauc> ${CMAKE_SOURCE_DIR})
  add_test(NAME jau_package_aot COMMAND sh ${CMAKE_SOURCE_DIR}/tests/package_aot.sh $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX} $<TARGET_FILE:jauc> $<TARGET_FILE:jur> ${CMAKE_SOURCE_DIR})
  add_test(NAME jau_jttp_http COMMAND sh ${CMAKE_SOURCE_DIR}/tests/jttp_http.sh $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX} $<TARGET_FILE:jauc> ${CMAKE_SOURCE_DIR})
endif()
add_test(NAME jau_runtime_api COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/runtime_api.jau -I ${CMAKE_SOURCE_DIR}/stdlib)
add_test(NAME jau_args COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/args.jau -- alpha beta)
add_test(NAME jaupm_where COMMAND jauc run ${CMAKE_SOURCE_DIR}/tools/jaupm.jau -I ${CMAKE_SOURCE_DIR}/stdlib -- where demo)
add_test(NAME jau_aot_functions COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/aot_functions.jau)
add_test(NAME jau_optimizer COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/optimizer.jau -O3)
add_test(NAME jau_arrays COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/arrays.jau -I ${CMAKE_SOURCE_DIR}/stdlib)
add_test(NAME jau_system_libs COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/system_libs.jau -I ${CMAKE_SOURCE_DIR}/stdlib)
add_test(NAME jau_time_api COMMAND jauc run ${CMAKE_SOURCE_DIR}/tests/time_api.jau)
install(TARGETS jauc jur jauas jauld jau-setup RUNTIME DESTINATION bin)
install(PROGRAMS $<TARGET_FILE_DIR:jauc>/jaupm${CMAKE_EXECUTABLE_SUFFIX} DESTINATION bin OPTIONAL)
install(FILES tools/jaupm.jau DESTINATION share/jau/tools)
install(DIRECTORY stdlib/ DESTINATION share/jau/stdlib)
